   294→          } catch {
   295→            // Non-critical: transient error (network blip, Supabase hiccup).
   296→            // The next heartbeat or online event will retry profile sync.
   297→          }
   298→        },

   314→                const cached = JSON.parse(raw);
   315→                if (cached?.access_token) {
   316→                  // Keep the authenticated state from bootstrapFromCache
   317→                  return;
   318→                }
   319→              }
   320→            } catch { /* ignore parse errors */ }
   321→          }
   322→          if (mounted) {
   323→            setProfile(null);
   324→            setStatus('unauthenticated');
   325→          }
   326→          return;
   327→        }
   328→
   329→        if (mounted) setAuth(session.user, session);
   330→
   331→        const metaCompleted = !!session.user.user_metadata?.profile_completed;
   332→        const profile = await getMyProfile();
   333→        if (!mounted) return;

   340→            fetchWatchlistFromServer(session.access_token);
   341→            // Bump updated_at to signal user activity (throttled to 1h)
   342→            touchProfile(profile?.updated_at);

   360→              if (!navigator.onLine) setOffline(true);

   367→          }
   368→        }
   369→      }
   370→    }

   372→    init();
   373→
   374→    return () => {
